cmake_minimum_required(VERSION 3.20)
project(pitchtracker_native LANGUAGES CXX)
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
find_package(pybind11 CONFIG REQUIRED)
option(PITCHTRACKER_CUDA "Build the optional CUDA residual evaluator" OFF)
pybind11_add_module(_pitchtracker_native bindings.cpp)
target_compile_features(_pitchtracker_native PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(_pitchtracker_native PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/fp:strict>)
else()
  target_compile_options(_pitchtracker_native PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-ffp-contract=off>)
endif()
if(PITCHTRACKER_CUDA)
  enable_language(CUDA)
  target_sources(_pitchtracker_native PRIVATE cuda_evaluator.cu)
  target_compile_definitions(_pitchtracker_native PRIVATE PITCHTRACKER_CUDA=1)
  set_target_properties(_pitchtracker_native PROPERTIES CUDA_STANDARD 17)
  target_compile_options(_pitchtracker_native PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
endif()
install(TARGETS _pitchtracker_native LIBRARY DESTINATION . RUNTIME DESTINATION .)
